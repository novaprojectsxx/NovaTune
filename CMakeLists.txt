cmake_minimum_required(VERSION 3.20)
project(Headroom VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(headroom_core native/core.cpp)
target_include_directories(headroom_core PUBLIC native)
if(MSVC)
  target_compile_options(headroom_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(headroom_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(headroom_tests tests/native/test_core.cpp)
  target_link_libraries(headroom_tests PRIVATE headroom_core)
  if(MINGW)
    target_link_options(headroom_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core COMMAND headroom_tests)
endif()
if(WIN32)
  add_library(headroom_windows native/windows.cpp native/network.cpp native/comparison.cpp native/experience.cpp native/tools.cpp native/boost.cpp native/input.cpp native/controller_art.cpp native/capture.cpp native/media_edit.cpp native/overlay.cpp native/desktop.cpp native/desktop_model.cpp native/desktop_shell.cpp native/desktop_media.cpp native/replay.cpp native/audio.cpp native/devices.cpp native/profiles.cpp native/maintenance.cpp native/timeline.cpp native/dialog.cpp native/fixes_dialog.cpp native/selection_dialog.cpp native/scrollbars.cpp native/gamer.cpp native/game_sessions.cpp native/stress.cpp native/uninstall.cpp native/registry.cpp native/storage.cpp native/system_tools.cpp native/session.cpp native/autoruns.cpp native/nvidia.cpp native/memory.cpp)
  target_compile_definitions(headroom_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(headroom_windows PUBLIC headroom_core comctl32 shell32 comdlg32 advapi32 psapi powrprof ole32 oleaut32 uuid wbemuuid dxgi version dwmapi uxtheme hid windowscodecs mfplat mfreadwrite mfuuid d3d11 d3dcompiler bcrypt crypt32 pdh iphlpapi ws2_32 setupapi cfgmgr32 taskschd winhttp msimg32 winmm)
  add_executable(Headroom WIN32 native/app.cpp native/Headroom.rc)
  target_link_libraries(Headroom PRIVATE headroom_windows)
  if(BUILD_TESTING)
    add_executable(headroom_fixture_game WIN32 tests/native/fixture_game.cpp)
    target_compile_definitions(headroom_fixture_game PRIVATE UNICODE _UNICODE)
    if(MINGW)
      target_link_options(headroom_fixture_game PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_executable(headroom_windows_tests tests/native/test_windows.cpp native/Headroom.rc)
    target_link_libraries(headroom_windows_tests PRIVATE headroom_windows)
    if(MSVC)
      target_link_options(headroom_windows_tests PRIVATE /MANIFEST:NO)
    endif()
    if(MINGW)
      target_link_options(headroom_windows_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME windows_integration COMMAND headroom_windows_tests)
    add_executable(novatune_folder_tests tests/native/test_folder_migration.cpp)
    target_link_libraries(novatune_folder_tests PRIVATE headroom_windows)
    if(MINGW)
      target_link_options(novatune_folder_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME folder_migration COMMAND novatune_folder_tests)
  endif()
  if(MINGW)
    target_link_options(Headroom PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    target_compile_options(Headroom PRIVATE /W4 /utf-8)
    target_link_options(Headroom PRIVATE /MANIFEST:NO)
    target_compile_options(headroom_windows PRIVATE /W4 /utf-8)
  endif()
  set_target_properties(Headroom PROPERTIES OUTPUT_NAME "NovaTune")
  install(TARGETS Headroom RUNTIME DESTINATION .)
endif()
